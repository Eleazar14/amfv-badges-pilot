/*******************************************************
 * PRUEBA PERSONAL ELEAZAR - Envío de constancia digital
 *
 * Hoja esperada:
 *   IA_FV_TEST_mailmerge
 *
 * Columnas esperadas:
 *   full_name, email, issued_on, credential_id,
 *   credential_url, assertion_url, linkedin_add_url,
 *   linkedin_share_url, evidence_url, certificate_url,
 *   sent, sent_at, message_id
 *
 * Esta prueba NO constituye envío oficial AMFV.
 *******************************************************/

const SHEET_NAME = "IA_FV_TEST_mailmerge";
const TEST_MODE = true;
const TEST_EMAIL = "eleazar.co14@gmail.com";
const MAX_SEND_PER_RUN = 1;

const EMAIL_SUBJECT_REAL =
  "[PRUEBA PERSONAL] Constancia digital de prueba";

const EMAIL_SUBJECT_TEST =
  "[PRUEBA PERSONAL] Constancia digital de prueba";

const SENDER_NAME = "Prueba personal Eleazar";

function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu("Prueba Constancias")
    .addItem("1. Revisar configuración", "checkSheetSetup")
    .addItem("2. Ver cuota de correo", "checkQuota")
    .addItem("3. Previsualizar primera fila pendiente", "previewFirstPendingEmail")
    .addItem("4. Enviar correo de prueba", "sendCredentialEmails")
    .addToUi();
}

function sendCredentialEmails() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME);

  if (!sheet) {
    throw new Error("No se encontró la pestaña: " + SHEET_NAME);
  }

  const data = sheet.getDataRange().getValues();

  if (data.length < 2) {
    throw new Error("La hoja no contiene filas de participantes.");
  }

  const headers = data[0].map(String);
  const col = buildColumnMap(headers);

  validateRequiredColumns(col);

  let sentCount = 0;
  let skippedCount = 0;

  for (let r = 1; r < data.length; r++) {
    const rowNumber = r + 1;
    const row = data[r];

    const fullName = getCell(row, col.full_name);
    const participantEmail = getCell(row, col.email);
    const issuedOn = getCell(row, col.issued_on);
    const credentialId = getCell(row, col.credential_id);
    const credentialUrl = getCell(row, col.credential_url);
    const assertionUrl = getCell(row, col.assertion_url);
    const linkedinAddUrl = getCell(row, col.linkedin_add_url);
    const linkedinShareUrl = getCell(row, col.linkedin_share_url);
    const evidenceUrl = getCell(row, col.evidence_url);
    const certificateUrl = getCell(row, col.certificate_url);
    const sentStatus = getCell(row, col.sent);

    if (String(sentStatus).toUpperCase() === "YES") {
      skippedCount++;
      continue;
    }

    if (!fullName || !participantEmail || !credentialId || !credentialUrl || !certificateUrl) {
      Logger.log("Fila " + rowNumber + " omitida por datos incompletos.");
      skippedCount++;
      continue;
    }

    const recipient = TEST_MODE ? TEST_EMAIL : participantEmail;
    const subject = TEST_MODE ? EMAIL_SUBJECT_TEST : EMAIL_SUBJECT_REAL;

    const htmlBody = buildHtmlEmail({
      fullName: fullName,
      participantEmail: participantEmail,
      issuedOn: issuedOn,
      credentialId: credentialId,
      credentialUrl: credentialUrl,
      assertionUrl: assertionUrl,
      linkedinAddUrl: linkedinAddUrl,
      linkedinShareUrl: linkedinShareUrl,
      evidenceUrl: evidenceUrl,
      certificateUrl: certificateUrl,
      testMode: TEST_MODE
    });

    const plainBody = buildPlainTextEmail({
      fullName: fullName,
      participantEmail: participantEmail,
      issuedOn: issuedOn,
      credentialId: credentialId,
      credentialUrl: credentialUrl,
      assertionUrl: assertionUrl,
      linkedinAddUrl: linkedinAddUrl,
      linkedinShareUrl: linkedinShareUrl,
      evidenceUrl: evidenceUrl,
      certificateUrl: certificateUrl,
      testMode: TEST_MODE
    });

    GmailApp.sendEmail(recipient, subject, plainBody, {
      htmlBody: htmlBody,
      name: SENDER_NAME
    });

    const now = new Date();
    sheet.getRange(rowNumber, col.sent_at + 1).setValue(now);

    if (TEST_MODE) {
      sheet.getRange(rowNumber, col.message_id + 1).setValue("TEST_SENT_TO_" + TEST_EMAIL);
    } else {
      sheet.getRange(rowNumber, col.sent + 1).setValue("YES");
      sheet.getRange(rowNumber, col.message_id + 1).setValue("SENT");
    }

    sentCount++;

    if (sentCount >= MAX_SEND_PER_RUN) {
      break;
    }
  }

  const modeText = TEST_MODE ? "MODO PRUEBA" : "MODO REAL";

  SpreadsheetApp.getUi().alert(
    modeText +
    "

Correos enviados en esta ejecución: " + sentCount +
    "
Filas omitidas o ya enviadas: " + skippedCount +
    "

TEST_MODE actual: " + TEST_MODE
  );
}

function checkSheetSetup() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME);

  if (!sheet) {
    SpreadsheetApp.getUi().alert("ERROR: No se encontró la pestaña: " + SHEET_NAME);
    return;
  }

  const data = sheet.getDataRange().getValues();
  const headers = data[0].map(String);
  const col = buildColumnMap(headers);
  const missing = getMissingRequiredColumns(col);

  if (missing.length > 0) {
    SpreadsheetApp.getUi().alert("ERROR: Faltan columnas requeridas:

" + missing.join("
"));
    return;
  }

  let totalRows = 0;
  let pendingRows = 0;
  let sentRows = 0;
  let incompleteRows = 0;

  for (let r = 1; r < data.length; r++) {
    const row = data[r];
    const fullName = getCell(row, col.full_name);
    const email = getCell(row, col.email);
    const credentialUrl = getCell(row, col.credential_url);
    const certificateUrl = getCell(row, col.certificate_url);
    const sent = getCell(row, col.sent);

    if (!fullName && !email && !credentialUrl && !certificateUrl) {
      continue;
    }

    totalRows++;

    if (String(sent).toUpperCase() === "YES") {
      sentRows++;
    } else {
      pendingRows++;
    }

    if (!fullName || !email || !credentialUrl || !certificateUrl) {
      incompleteRows++;
    }
  }

  SpreadsheetApp.getUi().alert(
    "Configuración revisada correctamente.

" +
    "Pestaña: " + SHEET_NAME + "
" +
    "TEST_MODE: " + TEST_MODE + "
" +
    "TEST_EMAIL: " + TEST_EMAIL + "

" +
    "Filas con datos: " + totalRows + "
" +
    "Pendientes: " + pendingRows + "
" +
    "Marcadas como enviadas: " + sentRows + "
" +
    "Filas incompletas: " + incompleteRows
  );
}

function buildColumnMap(headers) {
  const col = {};
  headers.forEach(function(header, index) {
    const key = String(header).trim();
    col[key] = index;
  });
  return col;
}

function getMissingRequiredColumns(col) {
  const required = [
    "full_name",
    "email",
    "issued_on",
    "credential_id",
    "credential_url",
    "assertion_url",
    "linkedin_add_url",
    "linkedin_share_url",
    "evidence_url",
    "certificate_url",
    "sent",
    "sent_at",
    "message_id"
  ];

  const missing = [];
  required.forEach(function(name) {
    if (col[name] === undefined) {
      missing.push(name);
    }
  });
  return missing;
}

function validateRequiredColumns(col) {
  const missing = getMissingRequiredColumns(col);
  if (missing.length > 0) {
    throw new Error("Faltan columnas requeridas:
" + missing.join("
"));
  }
}

function getCell(row, index) {
  if (index === undefined || index === null) {
    return "";
  }
  return String(row[index] || "").trim();
}

function escapeHtml(value) {
  return String(value || "")
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;");
}
